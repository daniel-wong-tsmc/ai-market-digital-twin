# F117 + F126 do-not-fetch registry — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** One registry file, `registry/do-not-fetch.json`, that records both publishers who
asked not to be used and sites that turn the verifier's plain reader away — enforced in the
fetch runner, the verifier and the researcher's brief, and learned automatically from
HTTP 401/403/429.

**Architecture:** A new stdlib-only leaf module `gpu_agent/fetch_policy.py` owns the shared
host matcher and the registry (load / match / idempotent append). Three consumers import it:
`gathering/webreach.py` (hard-refuses `publisher-objection`), `chartdata/verify.py`
(pre-rejects objections, classifies blocked fetch failures, auto-learns), and
`chartdata/research_prompt.py` (names both kinds in rule 8). The CLI grows a
`--do-not-fetch` flag on the two commands that need it.

**Tech Stack:** Python 3, stdlib only in the new module (`json`, `pathlib`, `urllib.parse`,
`urllib.error`), pytest.

```python
from __future__ import annotations

import json
import pathlib
from dataclasses import dataclass
from typing import Iterable
from urllib.parse import urlparse

DO_NOT_FETCH_REGISTRY = pathlib.Path("registry/do-not-fetch.json")

KIND_OBJECTION = "publisher-objection"
KIND_BLOCKS_READERS = "blocks-plain-readers"
KINDS = (KIND_OBJECTION, KIND_BLOCKS_READERS)

_LEARNED_WHY = ("learned automatically: the verifier's plain reader was turned "
                "away (HTTP 401/403/429)")
_KEY_ORDER = ("domain", "kind", "since", "why", "firstSeenUrl")


def matching_domain(target: str, domains: Iterable[str]) -> str | None:
    """The domain in `domains` that `target`'s host matches -- exact host or
    dot-suffix subdomain -- or None. Reads `parsed.hostname` so userinfo
    (`user:pass@host`) and a port cannot hide the real host, and returns None
    for anything that is not a fetchable http(s) URL."""
    try:
        parsed = urlparse(target)
    except ValueError:
        return None
    if parsed.scheme not in ("http", "https") or not parsed.netloc:
        return None
    host = (parsed.hostname or "").lower()
    if not host:
        return None
    for dom in sorted(d.lower() for d in domains):
        if host == dom or host.endswith("." + dom):
            return dom
    return None


@dataclass(frozen=True)
class DoNotFetchEntry:
    domain: str
    kind: str
    since: str
    why: str
    firstSeenUrl: str | None = None


class DoNotFetchRegistry:
    def __init__(self, entries: list[DoNotFetchEntry] | None = None) -> None:
        self.entries = sorted(entries or [], key=lambda e: e.domain)

    @property
    def is_empty(self) -> bool:
        return not self.entries

    def domains(self, kind: str | None = None) -> list[str]:
        return sorted(e.domain for e in self.entries if kind is None or e.kind == kind)

    def match(self, target: str, kind: str | None = None) -> DoNotFetchEntry | None:
        pool = [e for e in self.entries if kind is None or e.kind == kind]
        dom = matching_domain(target, [e.domain for e in pool])
        if dom is None:
            return None
        return next((e for e in pool if e.domain == dom), None)


def load_do_not_fetch(path=DO_NOT_FETCH_REGISTRY) -> DoNotFetchRegistry:
    """The registry at `path`, or an EMPTY registry when the file is missing,
    unreadable or malformed. Never raises: a registry that can strand a cycle
    is worse than no registry."""
    try:
        data = json.loads(pathlib.Path(path).read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return DoNotFetchRegistry()
    rows = data.get("entries") if isinstance(data, dict) else None
    if not isinstance(rows, list):
        return DoNotFetchRegistry()
    out = []
    for r in rows:
        if not isinstance(r, dict):
            continue
        domain = str(r.get("domain") or "").strip().lower()
        kind = r.get("kind")
        if not domain or kind not in KINDS:
            continue
        out.append(DoNotFetchEntry(
            domain=domain, kind=kind, since=str(r.get("since") or ""),
            why=str(r.get("why") or ""),
            firstSeenUrl=r.get("firstSeenUrl") or None))
    return DoNotFetchRegistry(out)


def record_blocked_domain(path, domain: str, *, since: str,
                          first_seen_url: str | None = None,
                          why: str | None = None) -> bool:
    """Append `domain` as a `blocks-plain-readers` entry. Returns False and
    changes nothing when the domain is already listed under EITHER kind (an
    objection must never be quietly downgraded to a technical block) or when
    the write fails -- a read-only checkout must not break a cycle."""
    domain = (domain or "").strip().lower()
    if not domain:
        return False
    path = pathlib.Path(path)
    reg = load_do_not_fetch(path)
    if any(e.domain == domain for e in reg.entries):
        return False
    rows = [{k: v for k, v in (
        ("domain", e.domain), ("kind", e.kind), ("since", e.since),
        ("why", e.why), ("firstSeenUrl", e.firstSeenUrl)) if k != "firstSeenUrl"
        or v is not None} for e in reg.entries]
    rows.append({"domain": domain, "kind": KIND_BLOCKS_READERS, "since": since,
                 "why": why or _LEARNED_WHY, "firstSeenUrl": first_seen_url})
    rows.sort(key=lambda r: r["domain"])
    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps({"version": 1, "entries": rows}, indent=2) + "\n",
                        encoding="utf-8", newline="\n")
    except OSError:
        return False
    return True
```

import urllib.error
from dataclasses import dataclass

from gpu_agent.fetch_policy import (
    DoNotFetchRegistry, KIND_BLOCKS_READERS, KIND_OBJECTION)

# HTTP answers that mean "this reader is not welcome" rather than "this page
# is gone" or "the network broke". F116 tail: the verifier used to report all
# three the same way, so nothing downstream could learn from a 403.
_BLOCKED_STATUSES = (401, 403, 429)


@dataclass(frozen=True)
class _FetchFailure:
    """Why one page could not be read, classified once and reused for every
    point that cites that page."""
    kind: str          # "blocked" | "not-found" | "unreachable"
    status: int | None
    detail: str


def _classify_fetch_failure(exc: Exception, known_blocker: bool) -> _FetchFailure:
    detail = f"{type(exc).__name__}: {exc}"
    status = getattr(exc, "code", None) if isinstance(exc, urllib.error.HTTPError) else None
    if status in _BLOCKED_STATUSES:
        return _FetchFailure("blocked", status, detail)
    if status == 404:
        return _FetchFailure("not-found", status, detail)
    if known_blocker:
        return _FetchFailure("blocked", None, detail)
    return _FetchFailure("unreachable", None, detail)


def _failure_line(i: int, url: str, f: _FetchFailure) -> str:
    if f.kind == "blocked" and f.status is not None:
        return f"point {i}: {url} blocked (HTTP {f.status})"
    if f.kind == "blocked":
        return (f"point {i}: {url} blocked (known to turn plain readers away; "
                f"{f.detail})")
    if f.kind == "not-found":
        return f"point {i}: {url} not found (HTTP {f.status})"
    return f"point {i}: {url} unreachable ({f.detail})"

def _reachability_rule(licensed: list[str], objections: list[str],
                       blocked: list[str]) -> str:
    """Rule 8. Three live cycles (2026-08-10/11) lost every candidate to a
    403 the brief never warned about; the 2026-08-19 cycle then lost one to a
    publisher NOT on any list, whose page opened cleanly for the researcher's
    own fetcher three times (F117). So the rule names the lists AND says
    plainly that checking it yourself proves nothing."""
    base = (
        "8. After you answer, a machine re-opens every URL you cite with a\n"
        "   plain automated reader (no login, no cookies, no browser) and\n"
        "   looks for each number on the page. That machine is a DIFFERENT\n"
        "   reader from the one you use: a page that opens cleanly for you\n"
        "   can still turn it away, so checking a page yourself proves\n"
        "   nothing about whether the number will verify. A site that turns\n"
        "   automated readers away (a paywall, a login wall, a \"403\n"
        "   Forbidden\" or \"access denied\" page, a bot check) can NEVER\n"
        "   pass that verification, however real its numbers are, and the\n"
        "   whole series is rejected. Prefer a publisher whose page opens\n"
        "   plainly."
    )
    if licensed:
        base += ("\n   Known licensed publishers that fail this check -- do not cite\n"
                 "   them as a point's source:\n"
                 + "\n".join(f"     - {d}" for d in licensed)
                 + "\n   If the only home for a number is one of these, treat it as\n"
                   "   unavailable.")
    if objections:
        base += ("\n   Publishers who have asked us not to use their material --\n"
                 "   NEVER cite these, whatever they publish:\n"
                 + "\n".join(f"     - {d}" for d in objections))
    if blocked:
        base += ("\n   Sites already known to turn the plain reader away -- treat\n"
                 "   anything they publish as unavailable, however well the page\n"
                 "   opens for you:\n"
                 + "\n".join(f"     - {d}" for d in blocked))
    return base
